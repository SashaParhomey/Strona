# Перенаправлення на index.php для красивих URL
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteBase /Strona samochod/
    RewriteRule ^index\.html$ - [L]
    RewriteCond %{REQUEST_FILENAME} !-f
    RewriteCond %{REQUEST_FILENAME} !-d
    RewriteRule . /Strona%20samochod/index.php [L]
</IfModule>

# Заборона доступу до файлів даних
<FilesMatch "\.json$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</FilesMatch>

# Заборона доступу до папки includes
<Directory "/xampp/htdocs/Strona samochod/includes">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
</Directory>
